type Factory<T> = () => T;

class LazyRegistry {
  private readonly cache = new Map<string, unknown>();

  // used ONLY by .use()
  public require<T>(key: string, factory: Factory<T>): T {
    const existing = this.cache.get(key) as T | undefined;
    if (existing) return existing;

    const created = factory();
    this.cache.set(key, created);
    return created;
  }

  // used by getters (no creation)
  public get<T>(key: string): T {
    // no if condition to mantain performance and if issue then it will give
    // and if we add if then we cna use require straight

  }
}
